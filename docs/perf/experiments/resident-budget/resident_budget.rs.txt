//! Regression controls for data-dependent resident sink growth.
//! Derived from the retained projection/aggregate paths and their callers.

use super::*;
use crate::exec::run::{Bindings, Sink as _};
use crate::work::{ExecutionPolicy, Resource, WorkContext, WorkError};

fn owned_bytes_policy(working_bytes: u64) -> ExecutionPolicy {
    ExecutionPolicy {
        input_bytes: u64::MAX,
        working_bytes,
        scratch_bytes: 0,
        result_bytes: u64::MAX,
        rows: u64::MAX,
        work_units: u64::MAX,
        timeout: std::time::Duration::from_secs(60),
    }
}

fn zero_owned_bytes() -> WorkContext {
    owned_bytes_policy(0).start().unwrap()
}

fn budget_for(work: &WorkContext) -> SinkBudget {
    SinkBudget {
        work: work.clone(),
        ram_bytes: crate::exec::scratch::DEFAULT_RAM_BYTES,
    }
}

fn assert_resource_refusal(error: crate::error::Error) {
    assert!(
        matches!(error, crate::error::Error::Store(ref store)
        if matches!(store.as_ref(), crate::storage::store::StoreError::Work(
            WorkError::Exhausted {
                resource: Resource::WorkingBytes | Resource::ScratchBytes,
                limit: 0,
                ..
            }
        ))),
        "expected typed owned-byte refusal, got {error:?}"
    );
}

fn full_posting_shape() -> (Schema, NormalizedQuery, ValidatedPlan) {
    let schema = schema();
    let normalized = normalized(
        &schema,
        vec![occurrence(0, POSTING, &[(0, 0), (1, 1), (2, 2)])],
        vec![],
    );
    let plan = planned(&schema, &normalized, &[0], &[0, 1, 2]);
    (schema, normalized, plan)
}

fn projection_set(unique: bool) -> SpillSet {
    let mut set = SpillSet::with_capacity_hint(3, 0, true);
    if unique {
        let (schema, normalized, _) = full_posting_shape();
        let finds = [0, 1, 2].map(|var| crate::ir::FindTerm::Var(VarId(var)));
        let witness = crate::plan::fj::provably_distinct_projection(&normalized, &schema, &finds)
            .expect("projecting a full relation row is proved unique");
        set.elide_output_hashing(witness);
    }
    set
}

fn scalar_projection_refuses_growth(unique: bool) {
    let mut set = projection_set(unique);
    let work = zero_owned_bytes();
    set.begin(Some(budget_for(&work)));
    for i in 0..512u64 {
        if !set.insert(&[i, i % 7, i + 1]) {
            break;
        }
    }
    assert_eq!(
        set.progress(),
        SinkProgress::Stop,
        "resident rows must not allocate beyond a zero working-byte allowance"
    );
    assert!(set.len() < 512);
    assert!(!set.insert(&[999, 8, 1000]), "refusal stays sticky");
    assert_resource_refusal(set.take_error().expect("typed refusal"));
    drop(set);
    assert_eq!(work.used(Resource::WorkingBytes), 0);
    assert_eq!(work.used(Resource::ScratchBytes), 0);
}

#[test]
fn resident_hashed_scalar_growth_obeys_working_budget() {
    scalar_projection_refuses_growth(false);
}

#[test]
fn resident_unique_scalar_growth_obeys_working_budget() {
    scalar_projection_refuses_growth(true);
}

#[test]
fn resident_hashed_bulk_growth_obeys_working_budget() {
    let mut set = projection_set(false);
    let work = zero_owned_bytes();
    set.begin(Some(budget_for(&work)));
    let words: Vec<u64> = (0..512).flat_map(|i| [i, i % 7, i + 1]).collect();
    set.insert_hashed_rows(&words);
    assert_eq!(set.progress(), SinkProgress::Stop);
    assert!(set.len() < 512, "bulk prefixes cannot bypass admission");
    assert_resource_refusal(set.take_error().expect("typed refusal"));
    drop(set);
    assert_eq!(work.used(Resource::WorkingBytes), 0);
}

#[test]
#[expect(
    unsafe_code,
    reason = "test writer initializes every proved-unique word"
)]
fn resident_unique_bulk_growth_obeys_working_budget() {
    let mut set = projection_set(true);
    let work = zero_owned_bytes();
    set.begin(Some(budget_for(&work)));
    let mut scratch = [0; 3];
    // SAFETY: every output word is initialized; each generated row has its
    // unique ordinal in the first column. No preexisting value is invalidated.
    unsafe {
        set.insert_unique_with(512, &mut scratch, |offset, words| {
            for (row, fields) in words.chunks_exact_mut(3).enumerate() {
                let i = u64::try_from(offset + row).unwrap();
                for (out, word) in fields.iter_mut().zip([i, i % 7, i + 1]) {
                    out.write(word);
                }
            }
        });
    }
    assert_eq!(set.progress(), SinkProgress::Stop);
    assert!(
        set.len() < 512,
        "unique direct-fill cannot bypass admission"
    );
    assert_resource_refusal(set.take_error().expect("typed refusal"));
    drop(set);
    assert_eq!(work.used(Resource::WorkingBytes), 0);
}

fn distinct_groups_refuse_growth(dense: bool) {
    let (_, _, plan) = full_posting_shape();
    let finds = [
        var_spec(&plan, 0),
        FindSpec::Agg(AggSpec::Count),
        agg_spec(&plan, FoldOp::Sum, 1, false),
    ];
    let mut sink = AggregateSink::without_seen_set(
        &finds,
        plan.slot_count(),
        plan.distinct_witness()
            .expect("full relation bindings are distinct"),
        0,
        if dense { &[4096] } else { &[] },
    );
    let work = zero_owned_bytes();
    sink.begin(Some(budget_for(&work)));
    let mut bindings = Bindings::new(plan.slot_count());
    for i in 0..512u64 {
        bindings.set(plan.slot_of(VarId(0)), i);
        bindings.set(plan.slot_of(VarId(1)), i % 7);
        bindings.set(plan.slot_of(VarId(2)), i + 1);
        sink.emit(&bindings);
        if sink.progress() != SinkProgress::Continue {
            break;
        }
    }
    assert_eq!(
        sink.progress(),
        SinkProgress::Stop,
        "group state needs admission even when binding dedup is lawfully elided"
    );
    let mut published = 0;
    let error = sink
        .finalize_into(&mut Vec::new(), |_| {
            published += 1;
            Ok(())
        })
        .expect_err("Q-ATOMIC refuses partial aggregate output");
    assert_resource_refusal(error);
    assert_eq!(published, 0);
    drop(sink);
    assert_eq!(work.used(Resource::WorkingBytes), 0);
    assert_eq!(work.used(Resource::ScratchBytes), 0);
}

fn resident_hashed_group_growth_obeys_working_budget() {
    distinct_groups_refuse_growth(false);
}

#[test]
fn resident_dense_group_growth_obeys_working_budget() {
    distinct_groups_refuse_growth(true);
}

#[test]
fn resident_hashed_scan_gather_obeys_working_budget() {
    use crate::exec::colt::SuffixRun;
    use crate::exec::run::{LeafScan, ScanOffer};

    let (schema, _, plan) = full_posting_shape();
    let facts: Vec<_> = (0..u64::from(STEP_QUANTUM))
        .map(|id| (id, id % 7, i64::try_from(id).unwrap()))
        .collect();
    let images = views_of(&schema, &facts, &[]);
    let colts = colts_for(&plan, &images);
    let key_slots: Vec<_> = plan.occurrences()[0].trie_schema[0]
        .iter()
        .map(|var| plan.slot_of(*var))
        .collect();
    let bindings = Bindings::new(plan.slot_count());
    let scan = LeafScan {
        colt: &colts[0],
        level: 0,
        key_slots: &key_slots,
        bindings: &bindings,
    };
    // The real executor delivers at most STEP_QUANTUM rows per window.
    // The small projected set does not pay for this separate gather buffer.
    let mut sink = ProjectionSink::new(vec![plan.slot_of(VarId(1))]);
    let initial_gather_capacity = sink.scan_rows.capacity();
    let work = zero_owned_bytes();
    sink.begin(Some(budget_for(&work)));
    sink.prepare_scan(&key_slots);
    assert_eq!(sink.begin_scan(&scan), ScanOffer::Open);
    sink.scan_run(
        &scan,
        SuffixRun::Identity {
            start: 0,
            len: usize::try_from(STEP_QUANTUM).unwrap(),
        },
    );
    assert_eq!(
        sink.scan_rows.capacity(),
        initial_gather_capacity,
        "gather must obtain admission before allocating, even if the set later refuses"
    );
    assert_eq!(sink.progress(), SinkProgress::Stop);
    assert_resource_refusal(sink.take_error().expect("typed refusal"));
    drop(sink);
    assert_eq!(work.used(Resource::WorkingBytes), 0);
}

#[test]
fn resident_exact_float_bank_obeys_working_budget_with_preexisting_other_banks() {
    let schema = SchemaDescriptor {
        relations: vec![RelationDescriptor {
            extension: None,
            name: "FloatBudget".into(),
            fields: [
                ("id", ValueType::U64),
                ("group", ValueType::U64),
                ("value", ValueType::F64),
            ]
            .into_iter()
            .map(|(name, value_type)| FieldDescriptor {
                name: name.into(),
                value_type,
            })
            .collect(),
        }],
        statements: vec![],
    }
    .validate()
    .unwrap();
    let normalized = normalized(
        &schema,
        vec![occurrence(0, POSTING, &[(0, 0), (1, 1), (2, 2)])],
        vec![],
    );
    let plan = planned(&schema, &normalized, &[0], &[0, 1, 2]);
    let finds = [
        var_spec(&plan, 1),
        FindSpec::Agg(AggSpec::Float {
            op: FoldOp::Sum,
            slot: plan.slot_of(VarId(2)),
        }),
        FindSpec::Agg(AggSpec::Float {
            op: FoldOp::Mean,
            slot: plan.slot_of(VarId(2)),
        }),
        FindSpec::Agg(AggSpec::Count),
    ];
    let mut sink = AggregateSink::without_seen_set(
        &finds,
        plan.slot_count(),
        plan.distinct_witness().unwrap(),
        0,
        &[4096],
    );
    // Isolate the float bank: these other capacities exist before binding.
    // No production charge is synthetically refunded or edited by this test.
    let GroupTable::Dense { ordinals, .. } = &mut sink.groups else {
        panic!("dense fixture");
    };
    ordinals.reserve(8);
    sink.group_counts.reserve(8);
    let GroupState::Folds { accs, .. } = &mut sink.group_state else {
        panic!("fold fixture");
    };
    accs.reserve(24);
    assert_eq!(sink.float_accs.capacity(), 0);
    let work = zero_owned_bytes();
    sink.begin(Some(budget_for(&work)));
    let mut bindings = Bindings::new(plan.slot_count());
    bindings.set(plan.slot_of(VarId(0)), 1);
    bindings.set(plan.slot_of(VarId(1)), 7);
    bindings.set(
        plan.slot_of(VarId(2)),
        bumbledb_theory::F64::from(1.0).to_order_key(),
    );
    sink.emit(&bindings);
    assert_eq!(sink.progress(), SinkProgress::Stop);
    assert_eq!(sink.float_accs.capacity(), 0, "refuse before allocating");
    let mut published = 0;
    let error = sink
        .finalize_into(&mut Vec::new(), |_| {
            published += 1;
            Ok(())
        })
        .expect_err("incomplete banks cannot publish a group");
    assert_resource_refusal(error);
    assert_eq!(published, 0);
    drop(sink);
    assert_eq!(work.used(Resource::WorkingBytes), 0);
}

#[test]
fn resident_pack_inner_growth_obeys_new_execution_budget() {
    let schema = schema();
    let normalized = normalized(
        &schema,
        vec![occurrence(0, PAYROLL, &[(0, 0), (1, 1), (2, 2)])],
        vec![],
    );
    let plan = planned(&schema, &normalized, &[0], &[0, 1, 2]);
    let finds = [
        var_spec(&plan, 1),
        FindSpec::Pack {
            slot: plan.slot_of(VarId(2)),
        },
    ];
    let mut sink =
        AggregateSink::new_distinct(&finds, plan.slot_count(), plan.distinct_witness().unwrap());
    let mut bindings = Bindings::new(plan.slot_count());
    let set_row = |bindings: &mut Bindings, id: u64| {
        bindings.set(plan.slot_of(VarId(0)), id);
        bindings.set(plan.slot_of(VarId(1)), 7);
        let slot = plan.slot_of(VarId(2));
        let start = i64::try_from(id * 2).unwrap();
        bindings.set(slot, i64_to_word(start));
        bindings.set(slot + 1, i64_to_word(start + 1));
    };
    // Construct the group and one inner capacity before the measured binding.
    set_row(&mut bindings, 0);
    sink.emit(&bindings);
    let GroupState::Pack { claims, .. } = &sink.group_state else {
        panic!("Pack fixture");
    };
    let initial_claim_capacity = claims[0].capacity();
    assert!(initial_claim_capacity > 0);
    sink.reset();
    let work = zero_owned_bytes();
    sink.begin(Some(budget_for(&work)));
    for id in 0..=u64::try_from(initial_claim_capacity).unwrap() {
        set_row(&mut bindings, id);
        sink.emit(&bindings);
        if sink.progress() != SinkProgress::Continue {
            break;
        }
    }
    assert_eq!(sink.progress(), SinkProgress::Stop);
    let GroupState::Pack { claims, .. } = &sink.group_state else {
        panic!("Pack fixture");
    };
    assert_eq!(claims[0].capacity(), initial_claim_capacity);
    let mut published = 0;
    let error = sink
        .finalize_into(&mut Vec::new(), |_| {
            published += 1;
            Ok(())
        })
        .expect_err("a refused Pack partition cannot publish partial intervals");
    assert_resource_refusal(error);
    assert_eq!(published, 0);
    drop(sink);
    assert_eq!(work.used(Resource::WorkingBytes), 0);
}

fn retained_projection_charges_survive_rebinding(unique: bool) {
    let policy = owned_bytes_policy(2 << 20);
    let first = policy.start().unwrap();
    let second = policy.start().unwrap();
    let mut set = projection_set(unique);
    set.begin(Some(budget_for(&first)));
    for i in 0..512u64 {
        assert!(set.insert(&[i, i % 7, i + 1]));
    }
    let retained = first.used(Resource::WorkingBytes);
    assert!(
        retained > 0,
        "data-dependent capacity has an owning reservation"
    );
    set.clear();
    assert_eq!(first.used(Resource::WorkingBytes), retained);
    set.begin(Some(budget_for(&second)));
    assert_eq!(
        first.used(Resource::WorkingBytes),
        retained,
        "rebind does not refund live storage"
    );
    for i in 0..128u64 {
        assert!(set.insert(&[i, i % 7, i + 1]));
    }
    assert_eq!(
        second.used(Resource::WorkingBytes),
        0,
        "reuse does not pay twice"
    );
    for i in 128..4096u64 {
        assert!(set.insert(&[i, i % 7, i + 1]));
    }
    assert!(
        second.used(Resource::WorkingBytes) > 0,
        "new growth belongs to the new operation"
    );
    // A WordMap replacement may legitimately release its old backing. A
    // retained vector growth may keep mixed-origin reservations. Do not force
    // identical refund timing onto these different ownership transitions.
    assert_eq!(set.progress(), SinkProgress::Continue);
    assert!(!set.spilled());
    drop(set);
    for work in [&first, &second] {
        assert_eq!(work.used(Resource::WorkingBytes), 0);
        assert_eq!(work.used(Resource::ScratchBytes), 0);
    }
}

fn resident_hashed_projection_retains_charges_across_clear_and_rebind() {

}

fn resident_unique_projection_retains_charges_across_clear_and_rebind() {

#[test]
fn resident_projection_owners_share_one_nonzero_allowance() {
    let work = owned_bytes_policy(4096).start().unwrap();
    let mut left = projection_set(true);
    let mut right = projection_set(true);
    for set in [&mut left, &mut right] {
        set.begin(Some(budget_for(&work)));
        assert!(set.insert(&[0, 0, 1]), "small seed fits each owner");
    }
    for i in 1..128u64 {
        left.insert(&[i, i % 7, i + 1]);
        right.insert(&[i, i % 7, i + 1]);
        if left.progress() != SinkProgress::Continue || right.progress() != SinkProgress::Continue {
            break;
        }
    }
    assert!(
        left.progress() == SinkProgress::Stop || right.progress() == SinkProgress::Stop,
        "two owners cannot each claim the full shared allowance"
    );
    let used = work.used(Resource::WorkingBytes);
    assert!(used > 0 && used <= 4096);
    assert!(left.len() + right.len() < 256);
    for set in [&mut left, &mut right] {
        if let Some(error) = set.take_error() {
            assert!(
                matches!(error, crate::error::Error::Store(ref store)
                if matches!(store.as_ref(), crate::storage::store::StoreError::Work(
                    WorkError::Exhausted { resource: Resource::WorkingBytes | Resource::ScratchBytes, .. }
                ))),
                "expected shared-budget refusal, got {error:?}"
            );
        }
    }
    drop(left);
    drop(right);
    assert_eq!(work.used(Resource::WorkingBytes), 0);
    assert_eq!(work.used(Resource::ScratchBytes), 0);
}
